        // Lista por defecto de 10 preguntas motivacionales y reflexivas

        // Cargar preguntas desde localStorage o cargar las predeterminadas

        function loadQuestions() {

            const saved = localStorage.getItem('daily_wheel_questions');

            if (saved) {

                try {

                    questions = JSON.parse(saved);

                } catch (e) {

                    questions = [...defaultQuestions];

                }

            } else {

                questions = [...defaultQuestions];

                saveQuestionsToStorage();

            }

            renderQuestionsList();

        }

        // Dibujar la ruleta en Canvas HTML5

        function drawWheel() {

            const numSlices = questions.length;

            const width = canvas.width;

            const height = canvas.height;

            const centerX = width / 2;

            const centerY = height / 2;

            const radius = width / 2 - 10;


            ctx.clearRect(0, 0, width, height);


            if (numSlices === 0) {

                // Dibujar ruleta vacía

                ctx.beginPath();

                ctx.arc(centerX, centerY, radius, 0, 2 * Math.PI);

                ctx.fillStyle = '#334155';

                ctx.fill();

                ctx.strokeStyle = '#475569';

                ctx.lineWidth = 4;

                ctx.stroke();


                ctx.fillStyle = '#94a3b8';

                ctx.font = 'bold 16px Plus Jakarta Sans';

                ctx.textAlign = 'center';

                ctx.textBaseline = 'middle';

                ctx.fillText('Agrega al menos 1 pregunta', centerX, centerY);

                return;

            }


            const sliceAngle = (2 * Math.PI) / numSlices;


            for (let i = 0; i < numSlices; i++) {

                const angle = i * sliceAngle;

                

                // Sector circular

                ctx.beginPath();

                ctx.moveTo(centerX, centerY);

                ctx.arc(centerX, centerY, radius, angle, angle + sliceAngle);

                ctx.closePath();

                ctx.fillStyle = sliceColors[i % sliceColors.length];

                ctx.fill();

                ctx.lineWidth = 2;

                ctx.strokeStyle = '#0f172a';

                ctx.stroke();


                // Renderizar Texto en el sector

                ctx.save();

                ctx.translate(centerX, centerY);

                ctx.rotate(angle + sliceAngle / 2);

                ctx.textAlign = 'right';

                ctx.textBaseline = 'middle';

                ctx.fillStyle = '#ffffff';

                ctx.font = 'bold 12px Plus Jakarta Sans';


                // Truncar texto si es muy largo

                let text = questions[i];

                if (text.length > 22) {

                    text = text.substring(0, 20) + '...';

                }


                ctx.fillText(text, radius - 20, 0);

                ctx.restore();

            }


            // Borde exterior estilizado

            ctx.beginPath();

            ctx.arc(centerX, centerY, radius, 0, 2 * Math.PI);

            ctx.lineWidth = 6;

            ctx.strokeStyle = '#334155';

            ctx.stroke();

        }


        // Ejecutar el giro de la ruleta

        function spinWheel() {

            if (isSpinning) return;


            const today = new Date().toDateString();

            const lastSpinDate = localStorage.getItem('daily_wheel_last_spin');


            if (lastSpinDate === today) {

                showToast('Ya realizaste tu giro de hoy. ¡Vuelve mañana!', 'warning');

                return;

            }


            if (questions.length === 0) {

                showToast('Por favor, añade al menos una pregunta a la lista.', 'error');

                return;

            }


            isSpinning = true;

            spinBtn.disabled = true;

            spinIcon.classList.add('fa-spin');

            resultCard.classList.add('hidden');


            // Selección aleatoria del ganador

            const numSlices = questions.length;

            const winnerIndex = Math.floor(Math.random() * numSlices);

            

            // Cálculo del ángulo objetivo (Puntero está arriba = -90° ó 270°)

            const sliceAngleDeg = 360 / numSlices;

            // Alinea el centro del sector con la parte superior

            const targetSliceCenter = (winnerIndex * sliceAngleDeg) + (sliceAngleDeg / 2);

            const targetRotationNeeded = 360 - targetSliceCenter - 90;


            // Múltiples vueltas completas (5 a 8 vueltas) para dar emoción

            const extraSpins = (5 + Math.floor(Math.random() * 3)) * 360;

            

            // Ajustar rotación actual manteniendo el acumulado

            const currentDegreeMod = currentRotation % 360;

            const rotationOffset = (targetRotationNeeded - currentDegreeMod + 360) % 360;

            

            currentRotation += extraSpins + rotationOffset;


            // Animación CSS por transform

            canvas.style.transform = `rotate(${currentRotation}deg)`;


            // Finalización de la animación (4 segundos)

            setTimeout(() => {

                isSpinning = false;

                spinIcon.classList.remove('fa-spin');


                const winningQuestion = questions[winnerIndex];

                

                // Mostrar resultado

                questionResult.textContent = winningQuestion;

                resultCard.classList.remove('hidden');


                // Guardar intento diario

                localStorage.setItem('daily_wheel_last_spin', today);

                localStorage.setItem('daily_wheel_today_question', winningQuestion);


                updateSpinButtonState(true, winningQuestion);

                showToast('¡Tu pregunta del día está lista!', 'success');

            }, 4000);

        }


        // Verificar el estado de tirada diaria

        function checkDailySpin() {

            const today = new Date().toDateString();

            const lastSpinDate = localStorage.getItem('daily_wheel_last_spin');

            const savedTodayQuestion = localStorage.getItem('daily_wheel_today_question');


            if (lastSpinDate === today && savedTodayQuestion) {

                updateSpinButtonState(true, savedTodayQuestion);

                questionResult.textContent = savedTodayQuestion;

                resultCard.classList.remove('hidden');

            } else {

                updateSpinButtonState(false);

            }

        }


        // Actualiza interfaz según si ya se giró hoy o no

        function updateSpinButtonState(hasSpunToday, questionText = '') {

            if (hasSpunToday) {

                spinBtn.disabled = true;

                spinBtnText.textContent = 'Giro Realizado Hoy';

                spinIcon.className = 'fas fa-check-circle text-emerald-400';

                spinStatusText.textContent = '¡Ya descubriste tu pregunta de hoy! Vuelve mañana.';

                spinStatusText.className = 'text-xs text-emerald-400 font-semibold mt-2';

            } else {

                spinBtn.disabled = false;

                spinBtnText.textContent = '¡Girar Ruleta!';

                spinIcon.className = 'fas fa-sync-alt text-xl';

                spinStatusText.textContent = 'Queda 1 giro disponible para hoy';

                spinStatusText.className = 'text-xs text-slate-400 mt-2 font-medium';

            }

        }


        // Renderizar panel de preguntas

        function renderQuestionsList() {

            questionsList.innerHTML = '';

            questionCount.textContent = `${questions.length} preguntas`;


            questions.forEach((q, index) => {

                const item = document.createElement('div');

                item.className = 'flex items-center justify-between bg-slate-800/50 border border-slate-700/60 rounded-xl p-2.5 text-xs text-slate-200 group hover:border-slate-600 transition';

                

                item.innerHTML = `

                    <div class="flex items-center gap-2 overflow-hidden mr-2">

                        <span class="w-2 h-2 rounded-full flex-shrink-0" style="background-color: ${sliceColors[index % sliceColors.length]}"></span>

                        <span class="truncate font-medium">${escapeHtml(q)}</span>

                    </div>

                    <div class="flex items-center gap-1 opacity-80 group-hover:opacity-100">

                        <button onclick="editQuestion(${index})" title="Editar" class="p-1.5 text-slate-400 hover:text-indigo-400 rounded-lg hover:bg-slate-700 transition">

                            <i class="fas fa-pen"></i>

                        </button>

                        <button onclick="deleteQuestion(${index})" title="Eliminar" class="p-1.5 text-slate-400 hover:text-rose-400 rounded-lg hover:bg-slate-700 transition">

                            <i class="fas fa-trash-can"></i>

                        </button>

                    </div>

                `;

                questionsList.appendChild(item);

            });

        }


        // Agregar una nueva pregunta

        function addQuestion(e) {

            e.preventDefault();

            const input = document.getElementById('newQuestionInput');

            const text = input.value.trim();


            if (!text) return;


            questions.push(text);

            saveQuestionsToStorage();

            renderQuestionsList();

            drawWheel();


            input.value = '';

            showToast('Pregunta agregada a la ruleta.', 'success');

        }


        // Editar una pregunta

        function editQuestion(index) {

            const currentText = questions[index];

            const newText = prompt('Editar pregunta:', currentText);

            

            if (newText !== null && newText.trim() !== '') {

                questions[index] = newText.trim();

                saveQuestionsToStorage();

                renderQuestionsList();

                drawWheel();

                showToast('Pregunta actualizada.', 'info');

            }

        }


        // Eliminar una pregunta

        function deleteQuestion(index) {

            if (questions.length <= 1) {

                showToast('Debes mantener al menos 1 pregunta en la ruleta.', 'warning');

                return;

            }

            questions.splice(index, 1);

            saveQuestionsToStorage();

            renderQuestionsList();

            drawWheel();

            showToast('Pregunta eliminada.', 'info');

        }


        // Reiniciar intento diario (Modo Demo)

        function resetDailySpin() {

            localStorage.removeItem('daily_wheel_last_spin');

            localStorage.removeItem('daily_wheel_today_question');

            resultCard.classList.add('hidden');

            updateSpinButtonState(false);

            showToast('Modo Demo: Giro diario restablecido.', 'info');

        }


        // Restablecer preguntas por defecto

        function restoreDefaultQuestions() {

            questions = [...defaultQuestions];

            saveQuestionsToStorage();

            renderQuestionsList();

            drawWheel();

            showToast('Se restablecieron las 10 preguntas por defecto.', 'info');

        }


        // Helper para mostrar notificaciones estilo Toast

        function showToast(message, type = 'info') {

            const toast = document.getElementById('toast');

            const toastMessage = document.getElementById('toastMessage');

            const toastIcon = document.getElementById('toastIcon');


            toastMessage.textContent = message;


            if (type === 'success') {

                toastIcon.className = 'fas fa-check-circle text-emerald-400';

            } else if (type === 'warning') {

                toastIcon.className = 'fas fa-exclamation-triangle text-amber-400';

            } else if (type === 'error') {

                toastIcon.className = 'fas fa-times-circle text-rose-400';

            } else {

                toastIcon.className = 'fas fa-info-circle text-indigo-400';

            }


            toast.classList.remove('translate-x-full');

            

            setTimeout(() => {

                toast.classList.add('translate-x-full');

            }, 3000);

        }


        // Escape de caracteres HTML por seguridad

        function escapeHtml(text) {

            const div = document.createElement('div');

            div.textContent = text;

            return div.innerHTML;

        }
